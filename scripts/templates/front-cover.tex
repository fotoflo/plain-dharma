% Plain Dharma — XeLaTeX FRONT cover, generated to any trim.
%
% Companion to back-cover.tex. The designer's InDesign front (cover.jpg) is
% locked to 6×9; this template reproduces that layout natively in the brand
% fonts so the front can be generated at any trim (notably 5×8) without cropping
% the 6×9 raster. It pairs with the generated back cover: same cream field, same
% Garamond, same gold "stitched" stripe — here on the LEFT (spine) edge.
%
% generate-front-cover.ts substitutes the __TOKENS__ per target. __FONTS_DIR__
% becomes the absolute fonts dir; __SUN_IMG__ the prepared (desaturated, cream-
% flattened) watercolor sun; the geometry/stripe tokens shrink the page and
% inset the artwork to clear the print bleed.
\documentclass[__FONTSIZE__]{article}

\usepackage{fontspec}
\usepackage[paperwidth=__PAPER_W__,paperheight=__PAPER_H__,top=__M_TOP__,bottom=__M_BOT__,left=__M_LEFT__,right=__M_RIGHT__]{geometry}
\usepackage{xcolor}
\usepackage{graphicx}
\usepackage{tikz}
\usepackage{eso-pic}

% Brand palette — matches back-cover.tex / the site tokens.
\definecolor{cream}{HTML}{F5EFE0}
\definecolor{ink}{HTML}{2A2520}
\definecolor{muted}{HTML}{6B5F50}
\definecolor{saffron}{HTML}{F26C15}
\definecolor{gold}{HTML}{FBC608}

% Local Garamond Libre — same OTFs as the website / book PDF / back cover.
\setmainfont{GaramondLibre}[
  Path=__FONTS_DIR__/,
  Extension=.otf,
  UprightFont=*-Regular,
  ItalicFont=*-Italic,
  BoldFont=*-Bold,
  BoldItalicFont=*-BoldItalic,
  Ligatures=TeX,
  Numbers=OldStyle
]

\pagecolor{cream}
\color{ink}
\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% Gold stitched stripe on the LEFT (spine) edge + dashed saffron stitch, the
% mirror of the back cover's right-edge stripe. current-page node needs two
% passes. On print trims the band is widened so it still reads ~0.5in after the
% left bleed is guillotined off.
\newlength{\stripeW}\setlength{\stripeW}{__STRIPE_W__}
\newlength{\stitchX}\setlength{\stitchX}{__STITCH_X__}
\AddToShipoutPictureBG{%
  \begin{tikzpicture}[remember picture,overlay]
    \fill[gold]
      (current page.north west) rectangle ([xshift=\stripeW]current page.south west);
    \draw[saffron,line width=1pt,dash pattern=on 3.5pt off 3.2pt]
      ([xshift=\stitchX]current page.north west) -- ([xshift=\stitchX]current page.south west);
  \end{tikzpicture}%
}
