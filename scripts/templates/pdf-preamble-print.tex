% Plain Dharma — XeLaTeX preamble for the PRINT PDF (KDP Print upload).
%
% Differs from pdf-preamble.tex in three ways:
%   1. Twoside book class with mirrored margins and a binding gutter
%   2. PDF page = 5.25"×8.25" (5×8 trim + 0.125" bleed each side)
%   3. Hyperlinks render as plain ink (no color) — print can't follow them

% Garamond Libre — same family as the screen PDF.
\setmainfont{GaramondLibre}[
  Path=__FONTS_DIR__/,
  Extension=.otf,
  UprightFont=*-Regular,
  ItalicFont=*-Italic,
  BoldFont=*-Bold,
  BoldItalicFont=*-BoldItalic,
  Ligatures=TeX,
  Numbers=OldStyle
]

% Brand palette — same colors so chapter titles, drop epigraphs etc. match
% the screen edition even though the page color is up to the variant.
\definecolor{cream}{HTML}{F5EFE0}
\definecolor{ink}{HTML}{2A2520}
\definecolor{muted}{HTML}{6B5F50}
\definecolor{saffron}{HTML}{C7651C}

% Per-variant page color. The build script substitutes this token.
% `__PAGECOLOR_SETUP__` becomes `\pagecolor{cream}` for the color variant,
% or empty for the B&W variant (white paper).
__PAGECOLOR_SETUP__

\color{ink}
\linespread{1.18}

% Chapter heading — same as screen but slightly tighter to suit 5×8.
\usepackage{titlesec}
\titleformat{\chapter}[display]
  {\normalfont\LARGE\bfseries\color{ink}\raggedright\hyphenpenalty=10000}
  {}
  {0pt}
  {\LARGE}
\titlespacing*{\chapter}{0pt}{30pt}{24pt}

\titleformat{\section}
  {\normalfont\Large\bfseries\color{ink}\raggedright\hyphenpenalty=10000}
  {}
  {0pt}
  {}

% Blockquote (per-chapter drops): italic, muted, indented.
\renewenvironment{quote}{%
  \begin{list}{}{%
    \leftmargin=1.2em\rightmargin=1.2em\itemindent=0pt%
  }\item\raggedright\itshape\color{muted}}{%
  \end{list}}

% Cap images at 70% of the text width.
\usepackage{graphicx}
\setkeys{Gin}{width=0.7\linewidth,keepaspectratio}

% Pali diacritics fallback (same as screen PDF).
\usepackage{newunicodechar}
\newfontfamily\palifallback{STIX Two Text}
\newunicodechar{ṇ}{{\palifallback ṇ}}
\newunicodechar{ṭ}{{\palifallback ṭ}}

% Let long source URLs break across lines. The measure is only 3.75in and an
% Access to Insight path is far wider, so without this the URL runs off the
% page and gets clipped. Uses url.sty (already pulled in by hyperref) rather
% than xurl, which BasicTeX doesn't ship.
\usepackage{url}
\Urlmuskip=0mu plus 1mu\relax
\def\UrlBreaks{\do\/\do\-\do\.\do\_\do\?\do\&\do\=\do\#\do\:}

\usepackage{enumitem}
\setlist{itemsep=0.15em,topsep=0.3em}

\setlength{\parindent}{0pt}
\setlength{\parskip}{0.6em}

\hyphenation{Mindfulness Loving-Kindness Dharma Buddha Buddhist}

% Typesetting touch-ups from reader feedback on the printed paperback.
\makeatletter
% Widows and orphans: never leave one line of a paragraph alone at the top or
% bottom of a page.
\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000
% The twoside book class defaults to \flushbottom, which stretches the gaps
% between paragraphs to make every page end at the same height — the uneven
% spacing readers noticed. Let pages run short instead.
\raggedbottom
% 12pt on a 3.75in measure is ~55 characters a line; protrusion straightens the
% right edge and a little emergency stretch keeps loose lines out of the margin.
\usepackage{microtype}
\microtypesetup{protrusion=true,expansion=false}
\setlength{\emergencystretch}{2em}
% The one remaining blank page (verso before page 1, so the body opens on a
% recto) shouldn't carry a running head or folio — that reads as a misprint.
\renewcommand{\cleardoublepage}{%
  \clearpage
  \if@twoside
    \ifodd\c@page\else
      \null\thispagestyle{empty}\newpage
    \fi
  \fi}
\makeatother

% Page numbers in ONE place on every page: centred at the foot. The class put
% them top-outer on text pages and bottom-centre on chapter openers. The running
% head stays at the top, small italic caps, without a number.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\setlength{\headheight}{14pt}
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
\renewcommand{\sectionmark}[1]{\markright{#1}}
\fancyhead[CE]{\footnotesize\itshape PLAIN DHARMA}
\fancyhead[CO]{\footnotesize\itshape\MakeUppercase{\leftmark}}
\fancyfoot[C]{\footnotesize\thepage}
% Unnumbered chapters (Preface, Closing, Sources…) don't update the marks, so
% they kept the previous chapter's head. Set it from the starred chapter too.
\makeatletter
\let\pd@schapter\@schapter
\def\@schapter#1{\pd@schapter{#1}\markboth{#1}{}}
\makeatother
\fancypagestyle{plain}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \fancyfoot[C]{\footnotesize\thepage}%
}

% Print hyperlink convention: render link text as plain ink (no color).
% Pandoc still emits \href{} but the visual treatment is uniform.
\AtBeginDocument{%
  \hypersetup{
    colorlinks=false,
    pdfborder={0 0 0},
    % Pandoc leaves pdfauthor empty when `author` arrives via -V, so set the
    % full attribution here (see TITLE_PAGE_AUTHOR_TEX in lib/book-source.ts).
    pdfauthor={Gautama Buddha; translated by Claude Opus; edited by Alex Miller},
    pdfsubject={Six Foundational Buddhist Teachings in Plain Modern English},
    pdfkeywords={Buddhism, suttas, dharma, meditation, mindfulness},
    pdfcreator={pandoc + XeLaTeX (print edition)}
  }%
}
